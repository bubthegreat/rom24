```python
# tools/kbk_import/cparse.py
"""Minimal C-initializer parsing: comments, #defines, balanced-brace table extraction."""
import re


def strip_comments(text: str) -> str:
    out, i, n = [], 0, len(text)
    while i < n:
        c = text[i]
        if c == '"':
            j = i + 1
            while j < n and text[j] != '"':
                j += 2 if text[j] == "\\" else 1
            out.append(text[i : j + 1])
            i = j + 1
        elif text.startswith("/*", i):
            i = text.index("*/", i) + 2
        elif text.startswith("//", i):
            i = text.index("\n", i) if "\n" in text[i:] else n
        else:
            out.append(c)
            i += 1
    return "".join(out)


def parse_defines(text: str) -> dict:
    defines = {}
    for m in re.finditer(r"^[ \t]*#define[ \t]+(\w+)[ \t]+(.+?)[ \t]*$",
                         strip_comments(text), re.MULTILINE):
        defines[m.group(1)] = m.group(2).strip()
    return defines


def extract_initializer(text: str, name: str) -> str:
    text = strip_comments(text)
    m = re.search(re.escape(name) + r"\s*\[[^\]]*\]\s*=\s*\{", text)
    if not m:
        raise ValueError(f"initializer for {name!r} not found")
    start = m.end() - 1  # at the '{'
    depth, i = 0, start
    while i < len(text):
        c = text[i]
        if c == '"':
            i += 1
            while text[i] != '"':
                i += 2 if text[i] == "\\" else 1
        elif c == "{":
            depth += 1
        elif c == "}":
            depth -= 1
            if depth == 0:
                return text[start : i + 1]
        i += 1
    raise ValueError(f"unbalanced braces in {name!r}")


def parse_braces(block: str):
    """block starts at '{' and ends at matching '}'. Returns nested lists;
    C strings -> ('str', value) tuples; other tokens -> stripped strings."""
    pos = [1]  # skip opening '{'

    def parse_list():
        items, token = [], []

        def flush():
            tok = "".join(token).strip()
            token.clear()
            if tok:
                items.append(tok)

        while True:
            c = block[pos[0]]
            if c == '"':
                chars, j = [], pos[0] + 1
                while block[j] != '"':
                    if block[j] == "\\":
                        chars.append({"n": "\n", "r": "\r", "t": "\t"}.get(block[j + 1], block[j + 1]))
                        j += 2
                    else:
                        chars.append(block[j])
                        j += 1
                # adjacent C string literals concatenate
                if items and isinstance(items[-1], tuple) and not "".join(token).strip():
                    items[-1] = ("str", items[-1][1] + "".join(chars))
                else:
                    flush()
                    items.append(("str", "".join(chars)))
                pos[0] = j + 1
            elif c == "{":
                flush()
                pos[0] += 1
                items.append(parse_list())
            elif c == "}":
                flush()
                pos[0] += 1
                return items
            elif c == ",":
                flush()
                pos[0] += 1
            else:
                token.append(c)
                pos[0] += 1

    return parse_list()
```

```python
# tools/kbk_import/resolve.py
import ast
import re


class Resolver:
    def __init__(self, defines: dict):
        self.defines = defines

    def num(self, token: str) -> int:
        expr, seen = str(token), set()
        while True:
            names = set(re.findall(r"[A-Za-z_]\w*", expr)) - seen
            subst = {n: self.defines[n] for n in names if n in self.defines}
            if not subst:
                break
            for n, v in subst.items():
                expr = re.sub(rf"\b{n}\b", f"({v})", expr)
                seen.add(n)
        expr = re.sub(r"'(.)'", lambda m: str(ord(m.group(1))), expr)
        node = ast.parse(expr, mode="eval")
        for sub in ast.walk(node):
            if not isinstance(sub, (ast.Expression, ast.BinOp, ast.UnaryOp, ast.Constant,
                                    ast.BitOr, ast.BitAnd, ast.LShift, ast.RShift, ast.Add,
                                    ast.Sub, ast.Mult, ast.USub, ast.Invert)):
                raise ValueError(f"unsupported C expr: {token!r}")
        return eval(compile(node, "<cexpr>", "eval"))  # noqa: S307 - AST-whitelisted

    def value(self, token):
        if isinstance(token, tuple) and token[0] == "str":
            return token[1]
        if token == "TRUE":
            return True
        if token == "FALSE":
            return False
        return self.num(token)

    def slot(self, token: str) -> int:
        m = re.fullmatch(r"SLOT\s*\(\s*(-?\d+)\s*\)", token.strip())
        if not m:
            raise ValueError(f"not a SLOT(): {token!r}")
        return int(m.group(1))

    def gsn_name(self, token: str):
        m = re.fullmatch(r"&\s*gsn_(\w+)", str(token).strip())
        return m.group(1) if m else None

    def spell_name(self, token: str):
        tok = str(token).strip()
        if tok in ("NULL", "0", "spell_null"):
            return None
        m = re.fullmatch(r"spell_(\w+)", tok)
        return m.group(1).replace("_", " ") if m else None
```

```python
# tools/kbk_import/emit.py
"""Emit generated content modules. All emitters return python source text."""
import pprint

HEADER = "# GENERATED by tools/kbk_import — do not hand-edit.\n"

# constants the engine needs at runtime (loader math + level bounds)
RUNTIME_CONSTANTS = ["MAX_LEVEL", "LEVEL_HERO", "LEVEL_IMMORTAL", "WEIGHT_STANDARD",
                     "AC_PER_ONE_PERCENT_DECREASE_DAMAGE", "MAX_STATS"]


def _fmt(name, obj):
    return f"{name} = {pprint.pformat(obj, width=100, sort_dicts=False)}\n"


def emit_constants(defines):
    from tools.kbk_import.resolve import Resolver
    r = Resolver(defines)
    body = "".join(f"{n} = {r.num(n)}\n" for n in RUNTIME_CONSTANTS if n in defines)
    return HEADER + body


def emit_classes(entries, r):
    classes, order = {}, []
    for e in entries:
        name = r.value(e[0])
        order.append(name)
        classes[name] = {
            "name": name,
            "who_name": r.value(e[1]),
            "align": r.num(e[2]),
            "xpadd": r.num(e[3]),
            "weapon": r.num(e[4]),
            "guild_rooms": [r.num(v) for v in e[5]],
            "skill_adept": r.num(e[6]),
            "thac0_00": r.num(e[7]),
            "thac0_32": r.num(e[8]),
            "hp_min": r.num(e[9]),
            "hp_max": r.num(e[10]),
            "f_mana": r.value(e[11]),
            "base_group": r.value(e[12]),
            "default_group": r.value(e[13]),
            "ctype": r.num(e[14]),
        }
    return HEADER + _fmt("CLASSES", classes) + _fmt("CLASS_ORDER", order), order
```

```python
def _per_class(values, r, order):
    return {order[i]: r.num(v) for i, v in enumerate(values) if i < len(order)}


def emit_skills(entries, r, class_order):
    skills = {}
    for e in entries:
        name = r.value(e[0])
        if not name or name == "reserved":
            continue
        skills[name] = {
            "name": name,
            "skill_level": _per_class(e[1], r, class_order),
            "rating": _per_class(e[2], r, class_order),
            "spell_fun": r.spell_name(e[3]),
            "target": r.num(e[4]),
            "minimum_position": r.num(e[5]),
            "pgsn": r.gsn_name(e[6]),
            "slot": r.slot(e[7]),
            "min_mana": r.num(e[8]),
            "beats": r.num(e[9]),
            "noun_damage": r.value(e[10]),
            "msg_off": r.value(e[11]),
            "msg_obj": r.value(e[12]),
            "ctype": r.num(e[13]),
        }
    return HEADER + _fmt("SKILLS", skills)


def emit_groups(entries, r, class_order):
    groups = {}
    for e in entries:
        name = r.value(e[0])
        if not name:
            continue
        groups[name] = {
            "name": name,
            "rating": _per_class(e[1], r, class_order),
            "spells": [r.value(s) for s in e[2] if r.value(s)],
        }
    return HEADER + _fmt("GROUPS", groups)
```

```python
def emit_races(race_entries, pc_entries, r, class_order):
    races = {}
    for e in race_entries:
        name = r.value(e[0])
        races[name] = {"name": name, "pc_race": r.value(e[1]),
                       "act": r.num(e[2]), "aff": r.num(e[3]), "off": r.num(e[4]),
                       "imm": r.num(e[5]), "res": r.num(e[6]), "vuln": r.num(e[7]),
                       "form": r.num(e[8]), "parts": r.num(e[9])}
    pc_races = {}
    for e in pc_entries:
        name = r.value(e[0])
        if name.startswith("null"):
            continue
        pc_races[name] = {
            "name": name,
            "who_name": r.value(e[1]).strip(),
            "align": r.num(e[2]),
            "xpadd": r.num(e[3]),
            "skills": [r.value(s) for s in e[4] if r.value(s)],
            "classes": {class_order[i]: bool(r.num(v))
                        for i, v in enumerate(e[5]) if i < len(class_order)},
            "stats": [r.num(v) for v in e[6]],
            "max_stats": [r.num(v) for v in e[7]],
            "size": r.num(e[8]),
            "set_race": r.value(e[9]),
        }
    return HEADER + _fmt("RACES", races) + _fmt("PC_RACES", pc_races)


def emit_materials(text: str) -> str:
    lines = iter(text.splitlines())
    for line in lines:
        if line.split() and line.split()[0] == "MATS":
            break
    materials = {}
    for line in lines:
        line = line.strip()
        if not line or "~" not in line:
            continue
        name, rest = line.split("~", 1)
        name = name.strip()
        nums = [int(x) for x in rest.split()[:5]]
        materials[name] = dict(zip(
            ("prot_pierce", "prot_bash", "prot_slash", "prot_magic", "relative_weight"),
            nums))
        materials[name]["name"] = name
    return HEADER + _fmt("MATERIALS", materials)
```

```python
"""Repeatable KBK content import: kbk C sources -> generated python modules + areas."""
import argparse
import pathlib

from tools.kbk_import import cparse, emit
from tools.kbk_import.resolve import Resolver

from rom24.content.constants import *  # noqa: F401,F403

"""


def main():

    ap.add_argument("--kbk", type=pathlib.Path,
                    default=pathlib.Path("~/Development/kbk").expanduser())
    ap.add_argument("--repo", type=pathlib.Path, default=pathlib.Path("."))
```

```python
def _finalize_kbk_object(item):
    from rom24 import content

    if item.item_type == merc.ITEM_ARMOR:
        cap = 0.0
        for slot, factor in _AC_CAP_BY_SLOT:
            if slot in slots:
                cap = factor
        hero2 = 1.5 * content.LEVEL_HERO * content.LEVEL_HERO
        for i, prot in enumerate(("prot_pierce", "prot_bash", "prot_slash", "prot_magic")):
            slope = cap * (content.AC_PER_ONE_PERCENT_DECREASE_DAMAGE * mat[prot] / 100.0) / hero2

    recommend = 0.3

        if slot in slots:
            recommend = factor
    if item.item_type == merc.ITEM_WEAPON:
```

```python
def load_improgs(area):
    while True:
        area, letter = game_utils.read_letter(area)
        if letter == "E":
            return area
        if letter == "*":
            area, _ = game_utils.read_to_eol(area)
            continue
        templates = {"I": instance.item_templates, "M": instance.npc_templates,
                     "R": instance.room_templates}.get(letter)
        if templates is None:
            logger.warning("load_improgs: letter %s not *IMRE", letter)
            area, _ = game_utils.read_to_eol(area)
            continue
        area, vnum = game_utils.read_int(area)
        area, progtype = game_utils.read_word(area, False)
        area, progname = game_utils.read_word(area, False)
        area, _ = game_utils.read_to_eol(area)
        template = templates.get(vnum)
        if template is None:
            logger.warning("load_improgs: unknown vnum %d", vnum)
            continue
        if not hasattr(template, "improgs"):
            template.improgs = []
        template.improgs.append((progtype, progname))
    return area
```
